\subsection{Polær til rektangulær}
\[
    z = |z| \cdot \left(\cos(\arg(z)) + \sin(\arg(z)) \cdot i \right) \kildetag{Ligning 4.5}
\]

\[
    e^{it} = \cos(t) + i \sin(t) \kildetag{Euler's formel}
\]
Medfører:
\begin{center}
    $\cos(t) = \frac{e^{it} + e^{-it}}{2}$ \hfill
    $\sin(t) = \frac{e^{it} - e^{-it}}{2i}$ \hfill
    \kilde{Ligning 4.9}
\end{center}

Lad $n \in \mathbb{N}$, da gælder
\begin{align*}
    \cos(n t) &= \Re((\cos(t)+\sin(t)i)^n) \\
    \sin(n t) &= \Im((\cos(t)+\sin(t)i)^n)
\end{align*}
\kilde{Sætning 4.5.1 - DeMoivres Formler}



\subsection{Aritmetik}
\subsubsection{Rektangulær form}
$z_1 = a + bi$ og $z_2 = c + di$ hvor $a,b,c,d \in \mathbb{R}$ og konstant $k \in \mathbb{R}$
\begin{align*}
    z_1 + z_2 &= (a+c) + (b+d)i \\
    z_1 - z_2 &= (a-c) + (b-d)i \\
    \\
    z_1 \cdot k &= (a \cdot k) + (b \cdot k)i\\
    z_1 \cdot z_2 &= (a \cdot c - b \cdot d) + (a \cdot d + b \cdot c)i \\
    z_1 \cdot \overline{z_1} &= a^2 + b^2 
    \kildetag{Lemma 4.2.1} \\
    \\
    \frac{z_1}{z_2} &= \frac{(a \cdot c + b \cdot d) + (b \cdot c - a \cdot d)i}{c^2 + d^2} \\
    \frac{1}{z_1} &= \frac{a}{a^2 + b^2} - \frac{b}{a^2 + b^2}i 
    \kildetag{Ligning 4.3} \\
\end{align*}

\subsubsection{Modulus og argument}
Lad $z_1, z_2 \in \mathbb{C} \backslash \{ 0 \}$, så gælder:
\begin{align*}
    |z_1 \cdot z_2| &= |z_1| \cdot |z_2| \\
    \arg(z_1 \cdot z_2) &= \arg(z_1) + \arg(z_2) \\
    |z_1 / z_2| &= |z_1| / |z_2| \\
    \arg(z_1 / z_2) &= \arg(z_1) - \arg(z_2) \\
    |z^n| &= |z|^n \quad n \in \mathbb{Z} \\
    \arg(z^n) &= n \cdot \arg(z) \quad n \in \mathbb{Z}
\end{align*}
\kilde{Sætning 4.6.2}
\begin{align*}
    \Re(z) &= |z| \cdot \cos(\arg(z)) \kildetag{Ligning 4.4} \\
    \Im(z) &= |z| \cdot \sin(\arg(z)) \kildetag{Ligning 4.4} \\
    |z| &= \sqrt{\Re(z)^2 + \Im(z)^2} \kildetag{Ligning 4.6} \\
    z_3 \cdot z_4 &= |z_3 \cdot z_4| \cdot e^{i \cdot (\arg(z_3) + \arg(z_4))}
\end{align*}

\begin{center}
    $|e^z| = e^{\Re(z)}$ \qquad
    $\arg(e^{z}) = \Im(z)$
\end{center}

\kilde{Ligning 4.10}

\subsection{Kompleks Konjugering}

Det komplekst konjugerede af $z=a+bi$ er $\overline{z}=a-bi$.

For $n \in \mathbb{Z}$, $a \in \mathbb{R}$ og $z, z_1, z_2 \in \mathbb{C}$ gælder:

\begin{center}
    $\overline{\overline{z}} = z$ \hfill
    $\overline{z_1 + z_2} = \overline{z_1} + \overline{z_2}$ \hfill
    $\overline{z_1 \cdot z_2} = \overline{z_1} \cdot \overline{z_2}$ \hfill
    $\overline{\left( \frac{1}{z} \right)} = \frac{1}{(\overline{z})}$ \hfill
    $\overline{z^n} = (\overline{z})^n$ \hfill
\end{center}
\kilde{Lemma 5.3.1}

\begin{center}
    $\overline{e^z} = e^{\overline{z}}$ \hfill
    $\overline{e^{ia}} = e^{-ia}$ \hfill
    $\overline{z} = |z| \cdot e^{-i \cdot \arg(z)}$
\end{center}
\kilde{Lemma 5.3.2}

\begin{center}
    $(z) \cdot (\overline{z}) = {\Re(z)}^2 + {\Im(z)}^2$
