\section{Preliminaries}

\subsection{Adjoint}

Aka \textit{conjugate transpose}. For en matrix $\mtrix A \in \mathbb{C}^{m \times n}$, er adjoint defineret som
\[
    \mtrix A^* = \overline{\mtrix A^T}
\]
(Den komplekst konjugerede af hver indgang, så transponeret).


\subsection{Egenskaber for kvadratiske matricer}

\bemærk{
For reelle matricer er $\mtrix A^* = \mtrix A^T$
}

\begin{table}[h]
    \centering
    \renewcommand{\arraystretch}{1.5}

    \begin{tabular}{r@{$\;=\;$}l|l|l}
        \multicolumn{2}{l|}{Egenskab} & Reelt navn & Komplekst navn \\
        \hline
        $\mtrix A^*$ & $\mtrix A $ & Symmetrisk & Hermitian \\
        $\mtrix A^* \mtrix A$ & $\mtrix I$ & Ortogonal & Unitary \\
        $\mtrix A^* \mtrix A$ & $\mtrix A \mtrix A^*$ & \multicolumn{2}{l}{Normal} \\
        $\mtrix A^2$ & $\mtrix A$ & \multicolumn{2}{l}{Idempotent} \\
    \end{tabular}

    \vspace{2em}

    \textbf{Specielt for Hermitian}

    \begin{tabular}{l|l}
        Egenskab & Navn \\
        \hline
        $\forall \lambda \in \mtrix A : \lambda > 0$ & Positiv Definit \\
        $\forall \lambda \in \mtrix A : \lambda \ge 0$ & Positiv Semidefinit \\
        $\forall \lambda \in \mtrix A : \lambda < 0$ & Negativ Definit \\
        $\exists \lambda < 0 \land \exists \lambda > 0$ & Indefinit \\
        $\mtrix A^2 = \mtrix A$ & Ortogonal Projektion

    \end{tabular}
\end{table}

\begin{itemize}
    \item Afbildnings-matricer er idempotente
    \item Ikke-invertible matricer kaldes singulær.
\end{itemize}

For unitære matricer, se sektion \ref{sec:unitary-matricer}.

\[
\text{Symmetrisk} \subsetneq \text{Hermitian} \subsetneq \text{Normal}
\]

Alle egenværdier for en Hermitian matrix er reelle.
